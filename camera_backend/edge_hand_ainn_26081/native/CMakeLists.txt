cmake_minimum_required(VERSION 3.16)
project(nyx310_hand LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)
set(CMAKE_BUILD_RPATH_USE_ORIGIN TRUE)
set(CMAKE_INSTALL_RPATH "$ORIGIN")
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH FALSE)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk/3rdparty")
include_directories(include "${SDK}/librknn_api/include")
if(CMAKE_CROSSCOMPILING)
    include_directories("${SDK}/opencv/include" "${SDK}/jsoncpp/include")
    set(CV_LIBS "-Wl,--start-group" "${SDK}/opencv/lib/libopencv_calib3d.a" "${SDK}/opencv/lib/libopencv_imgproc.a" "${SDK}/opencv/lib/libopencv_core.a" "${SDK}/opencv/share/OpenCV/3rdparty/lib/libtegra_hal.a" "-Wl,--end-group")
    set(JSON_LIB "${SDK}/jsoncpp/lib/libjson.a")
    set(NPU_LIB "${SDK}/librknn_api/aarch64/librknnrt.so")
else()
    find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(JSON REQUIRED jsoncpp)
    include_directories(${OpenCV_INCLUDE_DIRS} ${JSON_INCLUDE_DIRS} "${JSON_INCLUDE_DIRS}/json")
    set(CV_LIBS ${OpenCV_LIBS})
    set(JSON_LIB ${JSON_LIBRARIES})
    add_library(rknn_mock SHARED tests/rknn_mock.cpp)
    set_target_properties(rknn_mock PROPERTIES CXX_VISIBILITY_PRESET default)
    target_compile_definitions(rknn_mock PRIVATE HOST_TEST=1)
    set(NPU_LIB rknn_mock)
endif()
add_library(alg SHARED src/hand_core.cpp src/alg.cpp)
target_compile_definitions(alg PRIVATE ALG_EXPORT_ON)
target_compile_options(alg PRIVATE -Wall -Wextra -Wno-misleading-indentation)
target_link_options(alg PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,-Bsymbolic" "-Wl,-z,defs")
target_link_libraries(alg PRIVATE ${CV_LIBS} ${JSON_LIB} ${NPU_LIB} pthread dl z rt)
if(NOT CMAKE_CROSSCOMPILING)
    add_library(replay SHARED tests/replay_bridge.cpp src/hand_core.cpp)
    target_link_libraries(replay PRIVATE alg ${CV_LIBS})
endif()
add_executable(core_probe tests/cross_core_probe.cpp src/hand_core.cpp)
target_link_libraries(core_probe PRIVATE ${CV_LIBS} pthread dl z rt)
